!! should format a table from one of its cells !!
a    b

| Name | Value |
|---|---|
| [|a|] | 1 |
| long name | 2 |

c    d

[expect]
a    b

[|| Name      | Value |
| --------- | ----- |
| a         | 1     |
| long name | 2     ||]

c    d

!! should format a table from its header !!
a    b

| [|Name|] | Value |
|:--|--:|
| a | 1 |

c    d

[expect]
a    b

| Name | Value |
[|| :--- | ----: |
| a    |     1 ||]

c    d

!! should format a table from its delimiter row !!
a    b

| Name | Value |
|:-[|-|]:|---|
| a | 1 |

c    d

[expect]
a    b

| Name | Value |
[|| :--: | ----- |
|  a   | 1     ||]

c    d

|---|---|
| a | 1 |
