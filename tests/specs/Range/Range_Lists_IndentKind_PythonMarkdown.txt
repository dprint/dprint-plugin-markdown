!! should format an item of a nested list indented by four spaces !!
- a    b
    - [|c    d|]
    - e    f
- g    h

[expect]
- a    b
    [|- c d|]
    - e    f
- g    h

!! should format the outer item when the nested list is indented by two spaces !!
- a    b
  - [|c    d|]
  - e    f
- g    h

[expect]
[|- a b
    - c d
    - e f|]
- g    h

!! should format an item of an ordered list !!
1. a    b
2. [|c    d|]

    e    f
3. g    h

[expect]
1. a    b
[|2. c d|]

    e    f
3. g    h
